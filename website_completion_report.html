Ⅲ. 시스템 구축 및 배포 현황

타 시스템 및 웹 서비스 영역과의 간섭을 방지하고 보안성을 극대화하기 위해 독립 전용 디렉토리 /home/www/kdc 에 인프라를 구축하고 Nginx Server Block을 적용하였습니다.

server {
    listen 80;
    server_name k-dc.kr www.k-dc.kr 121.125.61.114;

    root /home/www/kdc;
    index index.html;

    charset utf-8;

    location / {
        try_files $uri $uri/ /index.html;
    }

    # 격벽 구조 보안 설정 (.git, .env 등 숨김 파일 차단)
    location ~ /\. {
        deny all;
        access_log off;
        log_not_found off;
    }

    # 정적 자원 속도 최적화
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|hwp|pdf)$ {
        expires 7d;
        add_header Cache-Control "public, no-transform";
    }
}
